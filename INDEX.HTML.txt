    <script>
        // ------------------------------------------------------------------
        // ১. আপনার Firebase কনফিগারেশন আপডেট করুন
        // ------------------------------------------------------------------
        const firebaseConfig = {
            apiKey: "YOUR_API_KEY",
            authDomain: "YOUR_PROJECT_ID.firebaseapp.com",
            databaseURL: "https://YOUR_PROJECT_ID-default-rtdb.firebaseio.com",
            projectId: "YOUR_PROJECT_ID",
            storageBucket: "YOUR_PROJECT_ID.appspot.com",
            messagingSenderId: "YOUR_SENDER_ID",
            appId: "YOUR_APP_ID"
        };

        // Firebase ইনিশিয়াল্যিজেশন
        firebase.initializeApp(firebaseConfig);
        const db = firebase.database();
        const storage = firebase.storage();

        let selectedFile = null;
        let filesData = [];

        // App Load
        window.onload = function() {
            listenRealtimeFiles();
            setupQrCode();
        };

        // ২. Firebase Realtime Data Listening (স্বয়ংক্রিয় আপডেট)
        function listenRealtimeFiles() {
            db.ref('uploads').on('value', snapshot => {
                const data = snapshot.val();
                filesData = [];
                if (data) {
                    Object.keys(data).forEach(key => {
                        filesData.unshift({ id: key, ...data[key] });
                    });
                }
                renderFiles();
                updateStats();
            });
        }

        function renderFiles() {
            const grid = document.getElementById('fileGrid');
            const emptyState = document.getElementById('emptyState');

            grid.innerHTML = '';

            if (filesData.length === 0) {
                emptyState.classList.remove('hidden');
                return;
            } else {
                emptyState.classList.add('hidden');
            }

            filesData.forEach(file => {
                const card = document.createElement('div');
                card.className = "bg-white border border-slate-200 rounded-2xl overflow-hidden shadow-sm flex flex-col justify-between";
                const isImage = file.type.includes('image');
                const isPrinted = file.status === 'printed';

                card.innerHTML = `
                    <div>
                        <div class="h-44 bg-slate-100 relative flex items-center justify-center overflow-hidden border-b border-slate-100">
                            ${isImage 
                                ? `<img src="${file.url}" class="w-full h-full object-cover">`
                                : `<div class="text-rose-500 text-5xl"><i class="fa-solid fa-file-pdf"></i></div>`
                            }
                            <span class="absolute top-2.5 left-2.5 px-2.5 py-0.5 rounded-full text-[10px] font-bold ${isPrinted ? 'bg-emerald-500 text-white' : 'bg-amber-500 text-white'}">
                                ${isPrinted ? 'প্রিন্ট সম্পন্ন' : 'অপেক্ষমাণ'}
                            </span>
                        </div>
                        <div class="p-4 space-y-2">
                            <h4 class="font-bold text-sm text-slate-800 truncate">${file.fileName}</h4>
                            <div class="text-xs text-slate-600 bg-slate-50 p-2.5 rounded-xl space-y-1">
                                <p class="font-semibold"><i class="fa-solid fa-user text-indigo-500 mr-1"></i> ${file.customerName}</p>
                                ${file.customerPhone ? `<p><i class="fa-solid fa-phone mr-1 text-slate-400"></i> ${file.customerPhone}</p>` : ''}
                                ${file.note ? `<p class="text-indigo-600 font-medium border-t pt-1"><i class="fa-solid fa-note-sticky mr-1"></i> ${file.note}</p>` : ''}
                            </div>
                        </div>
                    </div>

                    <div class="p-3 bg-slate-50 border-t border-slate-100 flex items-center justify-between text-xs">
                        <button onclick="toggleStatus('${file.id}', '${file.status}')" class="px-2.5 py-1.5 rounded-lg border text-[11px] ${isPrinted ? 'border-amber-300 bg-amber-50 text-amber-700' : 'border-emerald-300 bg-emerald-50 text-emerald-700'}">
                            ${isPrinted ? 'পেন্ডিং করুন' : 'মার্ক ডান'}
                        </button>
                        <div class="flex items-center gap-1.5">
                            <a href="${file.url}" target="_blank" download="${file.fileName}" class="bg-indigo-600 text-white px-3 py-1.5 rounded-lg font-medium flex items-center gap-1">
                                <i class="fa-solid fa-download"></i> সেভ
                            </a>
                            <button onclick="directPrint('${file.url}')" class="bg-slate-800 text-white px-3 py-1.5 rounded-lg font-medium flex items-center gap-1">
                                <i class="fa-solid fa-print"></i> প্রিন্ট
                            </button>
                            <button onclick="deleteFile('${file.id}')" class="text-rose-500 hover:bg-rose-50 p-1.5 rounded-lg">
                                <i class="fa-solid fa-trash"></i>
                            </button>
                        </div>
                    </div>
                `;
                grid.appendChild(card);
            });
        }

        function handleFileSelect(event) {
            const file = event.target.files[0];
            if (!file) return;
            selectedFile = file;
            document.getElementById('fileNameDisplay').innerText = file.name;
            document.getElementById('uploadBtn').disabled = false;
        }

        // ৩. Firebase Cloud-এ ফাইল আপলোড
        function uploadToFirebase() {
            if (!selectedFile) return;

            const name = document.getElementById('custName').value.trim() || 'বেনামী গ্রাহক';
            const phone = document.getElementById('custPhone').value.trim();
            const note = document.getElementById('custNote').value.trim();

            const progressContainer = document.getElementById('uploadProgress');
            const progressBar = document.getElementById('progressBar');
            const progressPercent = document.getElementById('progressPercent');
            const uploadBtn = document.getElementById('uploadBtn');

            progressContainer.classList.remove('hidden');
            uploadBtn.disabled = true;

            const storageRef = storage.ref('customer_files/' + Date.now() + '_' + selectedFile.name);
            const uploadTask = storageRef.put(selectedFile);

            uploadTask.on('state_changed', 
                (snapshot) => {
                    const progress = (snapshot.bytesTransferred / snapshot.totalBytes) * 100;
                    progressBar.style.width = progress + '%';
                    progressPercent.innerText = Math.round(progress) + '%';
                }, 
                (error) => {
                    alert('আপলোড ব্যর্থ হয়েছে: ' + error.message);
                    uploadBtn.disabled = false;
                }, 
                () => {
                    uploadTask.snapshot.ref.getDownloadURL().then((downloadURL) => {
                        db.ref('uploads').push({
                            customerName: name,
                            customerPhone: phone,
                            note: note,
                            fileName: selectedFile.name,
                            url: downloadURL,
                            type: selectedFile.type,
                            status: 'pending',
                            timestamp: new Date().toLocaleTimeString()
                        }).then(() => {
                            alert('আপনার ফাইল সফলভাবে পৌঁছাল!');
                            resetForm();
                            switchView('admin');
                        });
                    });
                }
            );
        }

        function resetForm() {
            selectedFile = null;
            document.getElementById('fileInput').value = '';
            document.getElementById('custName').value = '';
            document.getElementById('custPhone').value = '';
            document.getElementById('custNote').value = '';
            document.getElementById('fileNameDisplay').innerText = 'ছবি বা PDF বেছে নিন';
            document.getElementById('uploadProgress').classList.add('hidden');
            document.getElementById('uploadBtn').disabled = true;
        }

        function toggleStatus(id, currentStatus) {
            const newStatus = currentStatus === 'printed' ? 'pending' : 'printed';
            db.ref('uploads/' + id).update({ status: newStatus });
        }

        function deleteFile(id) {
            if (confirm('ফাইলটি ডিলিট করতে চান?')) {
                db.ref('uploads/' + id).remove();
            }
        }

        function directPrint(url) {
            const printImg = document.getElementById('printImage');
            printImg.src = url;
            document.getElementById('printArea').classList.remove('hidden');
            setTimeout(() => {
                window.print();
                document.getElementById('printArea').classList.add('hidden');
            }, 500);
        }

        function updateStats() {
            document.getElementById('statTotal').innerText = filesData.length;
            document.getElementById('statPending').innerText = filesData.filter(f => f.status === 'pending').length;
            document.getElementById('statPrinted').innerText = filesData.filter(f => f.status === 'printed').length;
        }

        function switchView(view) {
            if (view === 'admin') {
                document.getElementById('adminView').classList.remove('hidden');
                document.getElementById('customerView').classList.add('hidden');
            } else {
                document.getElementById('adminView').classList.add('hidden');
                document.getElementById('customerView').classList.remove('hidden');
            }
        }

        function setupQrCode() {
            const qrcodeContainer = document.getElementById("qrcode");
            if (qrcodeContainer) {
                qrcodeContainer.innerHTML = "";
                new QRCode(qrcodeContainer, {
                    text: window.location.href,
                    width: 150,
                    height: 150
                });
            }
        }

        function openQrModal() { document.getElementById('qrModal').classList.remove('hidden'); }
        function closeQrModal() { document.getElementById('qrModal').classList.add('hidden'); }
    </script>
